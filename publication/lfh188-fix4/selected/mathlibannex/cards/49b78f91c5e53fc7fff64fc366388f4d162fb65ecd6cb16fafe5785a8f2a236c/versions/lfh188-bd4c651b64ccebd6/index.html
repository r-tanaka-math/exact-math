def ExistsNaimarkCounterexampleOfDensity (κ : Cardinal.{u}) : Prop :=
  ∃ (A : Type u) (iA : NonUnitalCStarAlgebra A),
    letI := iA
    ∃ (oA : PartialOrder A),
      letI := oA
      ∃ (sA : StarOrderedRing A) (nA : Nontrivial A),
        letI := sA
        letI := nA
        ∃ (H : Type u) (nH : NormedAddCommGroup H),
          letI := nH
          ∃ (iH : InnerProductSpace ℂ H),
            letI := iH
            ∃ (cH : CompleteSpace H),
              letI := cH
