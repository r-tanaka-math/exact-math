75/-- Distinct chosen classes admit no unitary intertwiner. -/
76theorem no_unitaryIntertwiner_selectedRepresentation (root : PureState A)
77    {i j : GNSClass A} (hij : i ≠ j) (e : SelectedGNS root i ≃ₗᵢ[ℂ] SelectedGNS root j) :
78    ¬ StarAlgHom.Intertwines (selectedRepresentation root i)
79      (selectedRepresentation root j)
80      (e : SelectedGNS root i →L[ℂ] SelectedGNS root j) := by
81  intro he
82  apply hij
83  apply representative_injective_on_classes root
84  refine ⟨e, ?_⟩
85  intro a x
86  have hx := congrArg
87    (fun T : SelectedGNS root i →L[ℂ] SelectedGNS root j ↦ T x) (he a)
88  simpa [selectedRepresentation, ContinuousLinearMap.comp_apply] using hx
