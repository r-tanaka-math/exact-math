13/-- The norm-closed unital star subalgebra generated by a set of operators. -/
14noncomputable def generated (S : Set (H →L[ℂ] H)) : StarSubalgebra ℂ (H →L[ℂ] H) :=
15  (StarAlgebra.adjoin ℂ S).topologicalClosure

17/-- Every nominated generator belongs to the generated C*-subalgebra. -/
18theorem subset_generated (S : Set (H →L[ℂ] H)) : S ⊆ generated S :=
19  fun _ hT ↦ StarSubalgebra.le_topologicalClosure _ (StarAlgebra.subset_adjoin ℂ S hT)

21/-- The generated star subalgebra is norm closed. -/
22theorem isClosed_generated (S : Set (H →L[ℂ] H)) :
